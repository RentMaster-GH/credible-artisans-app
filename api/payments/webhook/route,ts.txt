import { NextResponse } from 'next/server'

const BACKEND_URL = process.env.NEXT_PUBLIC_BACKEND_URL || 'http://localhost:3000'

export async function POST(request: Request) {
  try {
    const body = await request.json()

    // Paystack / Flutterwave event verification check
    const event = body.event || body.type

    if (event === 'charge.success' || event === 'successful') {
      const paymentData = body.data || body
      const amountPaid = (Number(paymentData.amount) || 0) / 100 // Convert pesewas/cents to GHS/USD
      const metadata = paymentData.metadata || {}

      const jobId = metadata.job_id || null
      const artisanId = metadata.artisan_id || null
      const reference = paymentData.reference || paymentData.tx_ref || `WEBHOOK_${Date.now()}`

      // Calculate 5% platform commission
      const commissionRate = 0.05
      const commissionEarned = amountPaid * commissionRate

      // Forward webhook log to custom backend if a server service token is configured,
      // or log cleanly so Paystack/Flutterwave receives a 200 OK acknowledgement.
      const serviceToken = process.env.BACKEND_SERVICE_TOKEN
      if (serviceToken) {
        const res = await fetch(`${BACKEND_URL.replace(/\/$/, '')}/data/commissions`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            Authorization: `Bearer ${serviceToken}`,
          },
          body: JSON.stringify({
            job_id: jobId,
            artisan_id: artisanId,
            amount_charged: amountPaid,
            commission_rate: commissionRate,
            commission_earned: commissionEarned,
            payment_reference: reference,
            status: 'paid',
          }),
        })

        if (!res.ok) {
          const errData = await res.json().catch(() => ({}))
          console.error('Error logging commission to custom backend:', errData.error || res.statusText)
          return NextResponse.json({ error: 'Backend insert failed' }, { status: 500 })
        }
      } else {
        console.log('Webhook verified (set BACKEND_SERVICE_TOKEN to persist):', {
          job_id: jobId,
          artisan_id: artisanId,
          amount_charged: amountPaid,
          commission_earned: commissionEarned,
          reference,
        })
      }
    }

    return NextResponse.json({ received: true }, { status: 200 })
  } catch (err: any) {
    console.error('Webhook error:', err.message)
    return NextResponse.json({ error: 'Webhook handler failed' }, { status: 400 })
  }
}
