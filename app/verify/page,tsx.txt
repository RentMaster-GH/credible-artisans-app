'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import { backend } from '@/lib/backend'
import Navbar from '@/components/Navbar'

export default function VerifyPage() {
  const [idType, setIdType] = useState('Ghana Card')
  const [file, setFile] = useState<File | null>(null)
  const [loading, setLoading] = useState(false)
  const [success, setSuccess] = useState(false)

  const router = useRouter()

  const fileToBase64 = (selectedFile: File): Promise<string> => {
    return new Promise((resolve, reject) => {
      const reader = new FileReader()
      reader.onload = () => resolve(reader.result as string)
      reader.onerror = (err) => reject(err)
      reader.readAsDataURL(selectedFile)
    })
  }

  const handleVerificationSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    if (!file) {
      alert('Please select an ID document image to upload.')
      return
    }

    if (file.size > 2 * 1024 * 1024) {
      alert('Please upload an image smaller than 2MB.')
      return
    }

    setLoading(true)
    setSuccess(false)

    try {
      const me = await backend.getMe()
      const userId = me.payload?.sub as string
      if (!userId) {
        alert('You must be signed in to submit verification.')
        router.push('/login')
        return
      }

      const dataUrl = await fileToBase64(file)

      await backend.create('verifications', {
        user_id: userId,
        id_type: idType,
        id_document_url: dataUrl,
        status: 'pending',
        submitted_at: new Date().toISOString(),
      })

      setSuccess(true)
      setFile(null)
    } catch (err: any) {
      alert(err.message || 'Failed to submit verification.')
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="min-h-screen bg-gray-50 pb-16">
      <Navbar />

      <div className="max-w-xl mx-auto px-4 pt-10">
        <div className="bg-white rounded-2xl border border-gray-200 p-8 shadow-sm">
          <h1 className="text-2xl font-extrabold text-gray-900 mb-2">Identity Verification</h1>
          <p className="text-gray-500 text-sm mb-6">
            Upload a valid government-issued ID (Ghana Card or Passport) to earn your verified badge and build client trust.
          </p>

          {success && (
            <div className="mb-6 p-4 bg-emerald-50 border border-emerald-200 text-emerald-700 text-sm rounded-xl">
              Verification document submitted successfully! Our team will review it shortly.
            </div>
          )}

          <form onSubmit={handleVerificationSubmit} className="space-y-4">
            <div>
              <label className="block text-xs font-medium text-gray-700 mb-1">ID Document Type</label>
              <select
                value={idType}
                onChange={(e) => setIdType(e.target.value)}
                className="w-full px-4 py-2.5 border border-gray-300 rounded-xl text-sm outline-none bg-white focus:ring-2 focus:ring-emerald-500"
              >
                <option value="Ghana Card">Ghana Card</option>
                <option value="Passport">Passport</option>
                <option value="Driver License">Driver's License</option>
              </select>
            </div>

            <div>
              <label className="block text-xs font-medium text-gray-700 mb-1">Upload ID Image / Photo (Max 2MB)</label>
              <input
                type="file"
                accept="image/*"
                required
                onChange={(e) => setFile(e.target.files?.[0] || null)}
                className="w-full text-sm text-gray-500 file:mr-4 file:py-2.5 file:px-4 file:rounded-xl file:border-0 file:text-xs file:font-semibold file:bg-emerald-50 file:text-emerald-700 hover:file:bg-emerald-100 transition"
              />
            </div>

            <button
              type="submit"
              disabled={loading}
              className="w-full bg-emerald-600 hover:bg-emerald-700 text-white font-medium py-2.5 rounded-xl text-sm shadow-sm transition disabled:opacity-50"
            >
              {loading ? 'Uploading & Submitting...' : 'Submit for Verification'}
            </button>
          </form>
        </div>
      </div>
    </div>
  )
}
